<script>


let cat = {

    name:"たま",

    level:1,

    xp:0,

    hunger:50,

    happy:50

};


function loadGame(){


    const save = localStorage.getItem("nyalife");


    if(save){

        cat = JSON.parse(save);

    }


    updateUI();

}


function saveGame(){

    localStorage.setItem(

        "nyalife",

        JSON.stringify(cat)

    );

}


function updateUI(){


    document.getElementById("displayName").textContent =

        cat.name;


    document.getElementById("level").textContent =

        cat.level;


    document.getElementById("xp").textContent =

        cat.xp;


    document.getElementById("hungerBar").style.width =

        cat.hunger + "%";


    document.getElementById("happyBar").style.width =

        cat.happy + "%";


    saveGame();

}


function showMessage(text){


    document.getElementById("message").textContent =

        text;

}


function gainXP(amount){


    cat.xp += amount;


    if(cat.xp >= 100){


        cat.level++;

        cat.xp -= 100;


        showMessage("🎉 レベルアップ！");

    }


    updateUI();

}


function feed(){


    cat.hunger = Math.min(

        100,

        cat.hunger + 20

    );


    gainXP(10);


    showMessage("😺 おいしいにゃ！");

}


function play(){


    cat.happy = Math.min(

        100,

        cat.happy + 20

    );


    gainXP(15);


    showMessage("😻 たのしいにゃ！");

}


function sleepCat(){


    cat.hunger = Math.max(

        0,

        cat.hunger - 10

    );


    cat.happy = Math.min(

        100,

        cat.happy + 10

    );


    gainXP(8);


    showMessage("💤 ぐっすり寝たにゃ");

}


function saveName(){


    const value =

        document.getElementById("catName").value;


    if(value.trim() !== ""){


        cat.name = value;


        updateUI();


        showMessage(

            "🐾 名前を変更しました"

        );

    }

}


function resetGame(){


    if(confirm("セーブデータを削除しますか？")){


        localStorage.removeItem("nyalife");


        location.reload();

    }

}


setInterval(()=>{


    cat.hunger =

        Math.max(0, cat.hunger - 1);


    cat.happy =

        Math.max(0, cat.happy - 1);


    updateUI();


},30000);


loadGame();


</script>
